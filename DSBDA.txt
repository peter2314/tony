import pandas as pd
import numpy as np
df= pd.DataFrame({
    'Name':['A','B','C','D','E'],
    'Marks':[85,90,None,40,30],
    'Attendence':[90,None,80,70,95]
})
df
df['Marks'].fillna(df['Marks'].mean(), inplace=True)
df['Attendence'].fillna(df['Attendence'].median(), inplace=True)
Q1 = df['Marks'].quantile(0.25)
Q3 = df['Marks'].quantile(0.75)
IQR = Q3 - Q1
lower = Q1 - 1.5 * IQR
upper = Q3 + 1.5 * IQR
outliers=df[(df['Marks'] < lower) | (df['Marks'] > upper)]
print("Outliers:\n",outliers)
df['Marks_norm'] = (df['Marks'] - df['Marks'].min()) / (df['Marks'].max() - df['Marks'].min())
df['Marks_log']=np.log(df['Marks'])
print(df)

import pandas as pd

from sklearn.linear_model import LinearRegression

df=pd.read_csv("BostonHousing.csv")
df
df.shape
X= df.drop('medv',axis=1)
Y=df['medv']
X_train,X_test,Y_train,Y_test = train_test_split(X,Y,test_size=0.2)
model=LinearRegression()
model.fit(X_train,Y_train)
Y_pred = model.predict(X_test)

df

Practical 06

import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import GaussianNB
from sklearn.metrics import accuracy_score,confusion_matrix,precision_score,recall_score
df = pd.read_csv("iris.csv")
df
X = df.drop('Species', axis=1)
Y = df['Species']
X_train, X_test, Y_train, Y_test = train_test_split(X, Y, test_size=0.2)
model=GaussianNB()
model.fit(X_train,Y_train)
Y_pred = model.predict(X_test)
print("Confusion  Matrix:\n", confusion_matrix(Y_test, Y_pred))
print("Accuracy:", accuracy_score(Y_test, Y_pred))
print("Error Rate:", 1 - accuracy_score(Y_test, Y_pred))
print("Precision:", precision_score(Y_test, Y_pred, average='macro'))
print("Recall:", recall_score(Y_test, Y_pred, average='macro'))


Practical 07

import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem import PorterStemmer, WordNetLemmatizer
from nltk import pos_tag
from sklearn.feature_extraction.text import TfidfVectorizer
nltk.download('averaged_perceptron_tagger_eng')

text=" Data science is amazing and powerfull"
tokens = word_tokenize(text)
print(tokens)
pos_tags = pos_tag(tokens)
print("\nPOS Tagging:\n", pos_tags)
stop_words = set(stopwords.words('english'))
filtered = [w for w in tokens if w.lower() not in stop_words]
print(filtered)
ps = PorterStemmer()
stemmed = [ps.stem(w) for w in filtered]
print(stemmed)
lm = WordNetLemmatizer()
lemmatized = [lm.lemmatize(w) for w in filtered]
print(lemmatized)
vectorizer = TfidfVectorizer()
tfidf = vectorizer.fit_transform([text])
print(tfidf.toarray())
